## Как записаны правки

Как в планах 0.5.x и 0.6.0: «`путь`:» — файл целиком; «В `путь` заменить функцию `НАЧАЛО` целиком:» — функция до
первой строки `}`; «В `путь` после функции `НАЧАЛО` добавить:»; «В `путь` заменить: … на: …» — фрагмент встречается
ровно один раз; «В `путь` добавить в конец:». `.bat` и `.ini` — CRLF на диске, в блоках — LF.

### Task 1: Запись кадра — данные раньше команды (0.5.2, мелочь 2a)

Каждый вызов `FrameRecord` клал команду, потом её данные. Если память кончится между ними (подмена под
`guarded`), останется команда без данных, и `execute` прочтёт за концом вектора. Теперь сначала данные, потом команда:
нехватка памяти оставляет разве что данные, которых не касается ни одна команда. Для проверки — отказ выделения
памяти в тестах (`tests/fail_alloc.*`: `operator new` тестовой программы — `malloc`, который по просьбе теста
отказывает).

```cpp
#pragma once

namespace t {

// While one lives, allocations fail: after `allowed` more of them, every one throws std::bad_alloc - a test sees what
// a structure is left with when memory runs out half-way (0.5.2 review 2). The test program's operator new is
// malloc, failing on request (fail_alloc.cpp).
class FailAllocations {
public:
    explicit FailAllocations(int allowed);
    ~FailAllocations();
    FailAllocations(const FailAllocations&) = delete;
    FailAllocations& operator=(const FailAllocations&) = delete;
};

}  // namespace t
```

`tests/fail_alloc.cpp`:
```cpp
#include "fail_alloc.hpp"

#include <cstdlib>
#include <new>

namespace {
int g_allowed = -1;  // allocations left before they fail; -1: they never do
}  // namespace

namespace t {
FailAllocations::FailAllocations(int allowed) { g_allowed = allowed; }
FailAllocations::~FailAllocations() { g_allowed = -1; }
}  // namespace t

void* operator new(std::size_t size) {
    if (g_allowed == 0) throw std::bad_alloc();
    if (g_allowed > 0) --g_allowed;
    if (void* p = std::malloc(size ? size : 1)) return p;
    throw std::bad_alloc();
}

void operator delete(void* p) noexcept { std::free(p); }
void operator delete(void* p, std::size_t) noexcept { std::free(p); }
```

```cpp
TEST(frame_record_out_of_memory_leaves_no_command_without_its_data) {
    // Each call adds its data, then its command (0.5.2 review 2a): memory running out between the two must leave no
    // command whose data is missing - execute would read past the end. Three commands of another kind leave the
    // command list room for a fourth, so the call allocates only for its own data.
    const uint32_t pixels[4] = {1, 2, 3, 4};
    const uint8_t vertices[32] = {};
    const auto survives = [](bool clears, int allowed, const std::function<void(FrameRecord&)>& add) {
        FrameRecord r;
        for (int i = 0; i < 3; ++i) {
            if (clears)
                r.clear(ClearCmd{});
            else
                r.viewport(ViewportCmd{});
        }
        bool threw = false;
        {
            t::FailAllocations fail(allowed);
            try {
                add(r);
            } catch (const std::bad_alloc&) {
                threw = true;
            }
        }
        for (const Command& c : r.commands()) {
            const size_t n = c.type == Cmd::Clear      ? r.clears().size()
                             : c.type == Cmd::Viewport ? r.viewports().size()
                             : c.type == Cmd::Draw     ? r.draws().size()
                             : c.type == Cmd::Upload   ? r.uploads().size()
                                                       : r.writes().size();
            if (c.index >= n) return false;
        }
        return threw && r.commands().size() == 3;
    };
    CHECK(survives(false, 0, [](FrameRecord& r) { r.clear(ClearCmd{}); }));
    CHECK(survives(true, 0, [](FrameRecord& r) { r.viewport(ViewportCmd{}); }));
    CHECK(survives(true, 1, [&](FrameRecord& r) { r.upload(7, 0, Rect{0, 0, 2, 2}, pixels); }));  // its pixels go in
    CHECK(survives(true, 1, [&](FrameRecord& r) { r.writeTarget(Rect{0, 0, 2, 2}, pixels); }));
    CHECK(survives(true, 1, [&](FrameRecord& r) {  // its vertices go in
        DrawCmd d;
        d.fvf = d3d7::FVF_TL1;
        r.draw(d, vertices, 1, nullptr, 0);
    }));
}
```

```cpp
    // uncounted, for the budget to subtract later (0.5.2 review 2c).
```

```cpp
std::string replacePersonal(const RegCopy& copy) {
    const std::wstring base = gameKeyPaths().personal;
    HKEY parent = nullptr;
    const size_t slash = base.find_last_of(L'\\');
    if (RegOpenKeyExW(HKEY_CURRENT_USER, base.substr(0, slash).c_str(), 0, KEY_ALL_ACCESS | KEY_WOW64_64KEY,
                      &parent) == ERROR_SUCCESS) {
        const LONG r = RegDeleteTreeW(parent, base.substr(slash + 1).c_str());
        RegCloseKey(parent);
        if (r != ERROR_SUCCESS && r != ERROR_FILE_NOT_FOUND) return failure("Ragnarok", r);
    }
    if (!copy.exists) return "";
    std::string failed;  // each key and value on its own (spec §4.5): the rest is written all the same
    const auto add = [&failed](const std::string& f) { failed += (failed.empty() ? "" : ", ") + f; };
    for (const auto& key : copy.keys) {
        HKEY k = nullptr;  // keys without values are made too
        const LONG r = RegCreateKeyExW(HKEY_CURRENT_USER, keyOf(key.first).c_str(), 0, nullptr, 0,
                                       KEY_SET_VALUE | KEY_WOW64_64KEY, nullptr, &k, nullptr);
        if (r != ERROR_SUCCESS) {
            add(failure(key.first.empty() ? "Ragnarok" : key.first, r));
            continue;
        }
        RegCloseKey(k);
        for (const auto& v : key.second) {
            const std::string e = writePersonalValue(RegWrite{key.first, v.first, settings::RegOp::Set, v.second});
            if (!e.empty()) add(e);
        }
    }
    return failed;
}
```

```cpp
std::string valueText(const RegValue& v) {
    if (const auto d = v.asDword()) return std::to_string(*d);
    if (v.isSz()) return v.text;
    static const char kHex[] = "0123456789abcdef";
    std::string out;
    for (const uint8_t b : v.bytes) {
        if (!out.empty()) out += ' ';
        out += kHex[b >> 4];
        out += kHex[b & 15];
    }
    return out;
}

std::vector<CopyDifference> copyDifferences(const GameRegistry& r, const std::vector<std::string>& names) {
    std::vector<CopyDifference> out;

        const RegValue* mine = r.personal.find("", name);
        const RegValue* common = r.common.find("", name);
        if (mine && common && *mine != *common) out.push_back({name, valueText(*mine), valueText(*common)});

    return out;
}
```

### Task 9: С клавиатуры — Alt+буква, порядок Tab, группы, подсказки серых, списки; «Применить» при наборе (0.6.0, M5, M6)

  букву слова, иначе любую свободную; буквы нижних кнопок окна («&Применить», «и&грать», «&Отменить», «За&крыть»)

- **Группы.** Каждый элемент начинает свою группу (`WS_GROUP`), кроме переключателей после первого: стрелки ходят

```cpp
#include "app/mnemonic.hpp"

namespace uagfx::app {
namespace {

// Lower case for the letters labels have: Latin, Cyrillic with the Ukrainian ones; anything else as it is.
wchar_t fold(wchar_t c) {
    if (c >= L'A' && c <= L'Z') return wchar_t(c + 32);
    if (c >= 0x0410 && c <= 0x042F) return wchar_t(c + 0x20);  // А-Я
    if (c >= 0x0400 && c <= 0x040F) return wchar_t(c + 0x50);  // Ё, Є, І, Ї...
    if (c == 0x0490) return wchar_t(0x0491);                    // Ґ
    return c;
}

bool letter(wchar_t c) {
    const wchar_t f = fold(c);
    return (f >= L'a' && f <= L'z') || (f >= L'0' && f <= L'9') || (f >= 0x0430 && f <= 0x045F) || f == 0x0491;
}

}  // namespace

std::wstring withMnemonic(const std::wstring& text, std::wstring& taken) {
    size_t at = std::wstring::npos;
    for (int pass = 0; pass < 2 && at == std::wstring::npos; ++pass)  // the first letter of a word, then any
        for (size_t i = 0; i < text.size(); ++i) {
            const bool starts = i == 0 || !letter(text[i - 1]);
            if (letter(text[i]) && (pass == 1 || starts) && taken.find(fold(text[i])) == std::wstring::npos) {
                at = i;
                break;
            }
        }
    std::wstring out;
    for (size_t i = 0; i < text.size(); ++i) {
        if (i == at) out += L'&';
        out += text[i];
        if (text[i] == L'&') out += L'&';
    }
    if (at != std::wstring::npos) taken += fold(text[at]);
    return out;
}

void takeMnemonic(const std::wstring& text, std::wstring& taken) {
    for (size_t i = 0; i + 1 < text.size(); ++i) {
        if (text[i] != L'&') continue;
        if (text[i + 1] != L'&') {
            taken += fold(text[i + 1]);
            return;
        }
        ++i;  // "&&" is an '&'
    }
}

}  // namespace uagfx::app
```

```python
def fail(msg):
    print("FAIL: " + msg)
    sys.exit(1)


def main(path):
    try:
        data = open(path, "rb").read()
    except OSError:
        fail(path + " not found")
    pe = struct.unpack_from("<I", data, 0x3C)[0]
    if data[pe:pe + 4] != b"PE\0\0":
        fail("not a PE file")
    machine, nsec = struct.unpack_from("<HH", data, pe + 4)
    if machine != 0x14C:
        fail("not a 32-bit (i386) program")
    optsize = struct.unpack_from("<H", data, pe + 20)[0]
    opt = pe + 24
    if struct.unpack_from("<H", data, opt)[0] != 0x10B:
        fail("not PE32")
    if struct.unpack_from("<H", data, opt + 68)[0] != 2:
        fail("not a GUI program (subsystem %d)" % struct.unpack_from("<H", data, opt + 68)[0])
    imp_rva = struct.unpack_from("<I", data, opt + 104)[0]
    sections = [struct.unpack_from("<8sIIII", data, opt + optsize + i * 40) for i in range(nsec)]

    def off(rva):
        for _, vsize, va, rsize, raw in sections:
            if va <= rva < va + max(vsize, rsize):
                return rva - va + raw
        fail("rva %#x is outside the sections" % rva)

    def cstr(rva):
        o = off(rva)
        return data[o:data.index(b"\0", o)].decode()

    listed_late = sorted(set().union(*ALLOWED.values()) & NOT_ON_WINDOWS_7)
    if listed_late:
        fail("the list holds functions Windows 7 does not have: " + ", ".join(listed_late))
    dlls, imports = [], []
    o = off(imp_rva)
    while struct.unpack_from("<I", data, o + 12)[0]:
        lookup, _, _, name, first = struct.unpack_from("<IIIII", data, o)
        dll = cstr(name).lower()
        dlls.append(dll)
        t = off(lookup or first)
        while True:
            entry = struct.unpack_from("<I", data, t)[0]
            if entry == 0:
                break
            if not entry & 0x80000000:
                imports.append((dll, cstr(entry + 2)))
            t += 4
        o += 20
    bad = [d for d in dlls if d not in ALLOWED]
    if bad:
        fail("unexpected imports: " + ", ".join(bad))
    unknown = sorted("%s!%s" % (d, f) for d, f in imports if f not in ALLOWED[d])
    if unknown:
        fail("imports functions not on the list: " + ", ".join(unknown) +
             " - check that Windows 7 has them, then add them to ALLOWED")
    for needed in (b"asInvoker", b"PerMonitorV2", b"Microsoft.Windows.Common-Controls"):
        if needed not in data:
            fail("the manifest has no " + needed.decode())
    print("OK: %s: GUI, %d functions from %s, asInvoker, per-monitor DPI" % (path, len(imports), " ".join(dlls)))
```
